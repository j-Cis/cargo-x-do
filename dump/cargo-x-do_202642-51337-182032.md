pub fn run_tui() {
	let has_history = last_and_history::has_history();

	loop {
		match run_level_1(has_history) {
			Ok(Some((action, is_dry_run))) => {
				let executed = match action {
					Action::Last => last_and_history::handle_last(is_dry_run),
					Action::LastList => last_and_history::handle_list(is_dry_run),
					Action::Info => info_and_save::handle_info(is_dry_run),
					Action::Run | Action::Build => run_and_build::handle_run_and_build(is_dry_run),
					Action::Git => git_and_cargo::handle_git(is_dry_run),
					Action::Cargo => git_and_cargo::handle_cargo(is_dry_run),
					Action::Setup => handle_setup(is_dry_run),
				};

				if executed {
					break;
				}
			}
			Ok(None) => break,
			Err(e) => {
				eprintln!("🚨 Błąd terminala: {}\r\n", e);
				break;
			}
		}
	}
}

pub fn run_level_1(has_history: bool) -> Result<Option<(Action, bool)>> {
	let mut out = stdout();
	terminal::enable_raw_mode()?;
	execute!(out, cursor::Hide)?;

	let mut is_dry_run = false;
	let mut first_draw = true;
	let mut final_result = None;

	'main_loop: loop {
		if !first_draw {
			execute!(
				out,
				cursor::MoveUp(10), // Podniesione do 10, bo doszła linia setup
				terminal::Clear(terminal::ClearType::FromCursorDown)
			)?;
		}
		first_draw = false;

		print!("\r\n🚀 x-do (xtask)\r\n\r\n");

		if has_history {
			print!("  {} ♻️  last      {} 📜 last list\r\n\r\n", "[Enter]".yellow().bold(), "[l]".yellow().bold());
		} else {
			print!(
				"  {} ♻️  last      {} 📜 last list  {}\r\n\r\n",
				"[Enter]".dark_grey(),
				"[l]".dark_grey(),
				"🚫".dark_grey()
			);
		}

		let dry_key = if is_dry_run { "[o]".red().bold() } else { "[o]".yellow().bold() };
		let dry_lbl = if is_dry_run { "🔒 dry: ON " } else { "🔓 dry: OFF" };

		print!(
			"  {} ℹ️  info      {} 🏗️  build      {} 🌿 git\r\n",
			"[i]".yellow().bold(),
			"[b]".yellow().bold(),
			"[g]".yellow().bold()
		);
		print!(
			"  {} ▶️  run       {} {}   {} 📦 cargo\r\n",
			"[r]".yellow().bold(),
			dry_key,
			dry_lbl,
			"[c]".yellow().bold()
		);
		print!("  {} ⚙️  setup\r\n\r\n", "[s]".yellow().bold());

		print!("❌  {}\r\n", "[q] 🔚 exit / [Ctrl+C]".dark_grey());

		out.flush()?;

		loop {
			if let Event::Key(key) = event::read()? {
				if key.kind != KeyEventKind::Press {
					continue;
				}

				if (key.modifiers.contains(KeyModifiers::CONTROL) && key.code == KeyCode::Char('c'))
					|| key.code == KeyCode::Esc
					|| key.code == KeyCode::Char('q')
				{
					break 'main_loop;
				}

				match key.code {
					KeyCode::Char('o') => {
						is_dry_run = !is_dry_run;
						break;
					}
					KeyCode::Enter if has_history => {
						final_result = Some((Action::Last, is_dry_run));
						break 'main_loop;
					}
					KeyCode::Char('l') if has_history => {
						final_result = Some((Action::LastList, is_dry_run));
						break 'main_loop;
					}
					KeyCode::Char('i') => {
						final_result = Some((Action::Info, is_dry_run));
						break 'main_loop;
					}
					KeyCode::Char('r') => {
						final_result = Some((Action::Run, is_dry_run));
						break 'main_loop;
					}
					KeyCode::Char('b') => {
						final_result = Some((Action::Build, is_dry_run));
						break 'main_loop;
					}
					KeyCode::Char('g') => {
						final_result = Some((Action::Git, is_dry_run));
						break 'main_loop;
					}
					KeyCode::Char('c') => {
						final_result = Some((Action::Cargo, is_dry_run));
						break 'main_loop;
					}
					KeyCode::Char('s') => {
						final_result = Some((Action::Setup, is_dry_run));
						break 'main_loop;
					}
					_ => {}
				}
			}
		}
	}

	terminal::disable_raw_mode()?;
	execute!(out, cursor::Show)?;
	print!("\r\n");

	Ok(final_result)
}

fn tag_release_with_hash(is_dry_run: bool) {
    let version = match read_cargo_version() {
        Some(v) => v,
        None => {
            eprintln!("🚨 Nie udało się odczytać wersji z Cargo.toml");
            return;
        }
    };

    let commit_msg = match inquire::Text::new("Wprowadź wiadomość commita wydania (np. 'docs: ...'):")
        .with_default(&format!("Release v{}", version))
        .prompt()
    {
        Ok(msg) => msg.trim().to_string(),
        Err(_) => {
            println!("Anulowano proces wydania.");
            return;
        }
    };

    if is_dry_run {
        println!("> [DRY RUN] 🔒 Wykonałbym pełen łańcuch wydania:");
        println!("  git add .");
        println!("  git commit -m \"{}\"", commit_msg);
        println!("  $hash = git rev-parse --short HEAD");
        println!("  git tag -a \"v{}-$hash\" -m \"Release v{} hash: $hash\"", version, version);
        println!("  git push origin main --follow-tags");
        println!("  git push origin --tags\r\n");
        return;
    }

    println!("📥 Dodawanie plików (git add .)...");
    let _ = Command::new("git").args(["add", "."]).status();

    println!("📝 Zapisywanie commita (git commit)...");
    let commit_status = Command::new("git")
        .args(["commit", "-m", &commit_msg])
        .status();

    if commit_status.is_err() || !commit_status.unwrap().success() {
        eprintln!("⚠️ Nie udało się utworzyć commita (być może nie ma żadnych zmian). Próbuję kontynuować tagowanie...");
    }

    let hash_out = Command::new("git").args(["rev-parse", "--short", "HEAD"]).output();
    let hash = match hash_out {
        Ok(out) if out.status.success() => {
            String::from_utf8_lossy(&out.stdout).trim().to_string()
        }
        _ => {
            eprintln!("🚨 Błąd pobierania hasha z git rev-parse");
            return;
        }
    };

    let tag_name = format!("v{}-{}", version, hash);
    let tag_msg = format!("Release v{} hash: {}", version, hash);

    println!("🏷️ Tworzenie taga: {}", tag_name);
    let status_tag = Command::new("git")
        .args(["tag", "-a", &tag_name, "-m", &tag_msg])
        .status();

    if let Ok(st) = status_tag {
        if st.success() {
            println!("🚀 Wysyłanie gałęzi głównej wraz z powiązanymi tagami (git push origin main --follow-tags)...");
            let _ = Command::new("git").args(["push", "origin", "main", "--follow-tags"]).status();

            println!("🚀 Weryfikacja wysłania wszystkich tagów (git push origin --tags)...");
            let _ = Command::new("git").args(["push", "origin", "--tags"]).status();

            crate::last_and_history::add_to_history("git", &["push", "origin", "main", "--follow-tags"]);
            crate::last_and_history::add_to_history("git", &["push", "origin", "--tags"]);
        } else {
            eprintln!("🚨 Nie udało się utworzyć taga (być może taki tag już istnieje).");
        }
    } else {
        eprintln!("🚨 Błąd wywołania komendy git tag.");
    }
}

fn bump_cargo_version(is_dry_run: bool) {
	let manifest_path = "Cargo.toml";
	let content = match fs::read_to_string(manifest_path) {
		Ok(c) => c,
		Err(_) => {
			eprintln!("🚨 Nie znaleziono pliku Cargo.toml");
			return;
		}
	};

	let current_version = match read_cargo_version() {
		Some(v) => v,
		None => {
			eprintln!("🚨 Nie znaleziono pola `version` w Cargo.toml");
			return;
		}
	};

	let suggested_version = suggest_next_version(&current_version);

	let new_version = match Text::new("Wprowadź nową wersję:").with_default(&suggested_version).prompt() {
		Ok(v) => v.trim().to_string(),
		Err(_) => {
			println!("Anulowano podbijanie wersji.");
			return;
		}
	};

	if new_version.is_empty() || new_version == current_version {
		println!("Wersja bez zmian.");
		return;
	}

	if is_dry_run {
		println!("> [DRY RUN] 🔒 Podmieniłbym w Cargo.toml: `{}` -> `{}`\r\n", current_version, new_version);
		return;
	}

	// Bezpieczna podmiana wyłącznie linii w sekcji [package]
	let mut replaced = false;
	let mut new_lines = Vec::new();

	for line in content.lines() {
		if !replaced
			&& line.trim().starts_with("version")
			&& let Some((k, _)) = line.split_once('=')
		{
			new_lines.push(format!("{}= \"{}\"", k, new_version));
			replaced = true;
		} else {
			new_lines.push(line.to_string());
		}
	}

	let mut output = new_lines.join("\n");
	if content.ends_with('\n') {
		output.push('\n');
	}

	match fs::write(manifest_path, output) {
		Ok(_) => println!("✅ Zaktualizowano Cargo.toml do wersji: {}\r\n", new_version),
		Err(e) => eprintln!("🚨 Błąd zapisu do Cargo.toml: {}\r\n", e),
	}
}

fn read_cargo_version() -> Option<String> {
	let content = fs::read_to_string("Cargo.toml").ok()?;
	for line in content.lines() {
		let trimmed = line.trim();
		if trimmed.starts_with("version")
			&& let Some((_, val)) = trimmed.split_once('=')
		{
			let clean = val.trim().trim_matches('"').trim_matches('\'');
			if !clean.is_empty() {
				return Some(clean.to_string());
			}
		}
	}
	None
}

fn suggest_next_version(ver: &str) -> String {
	// 1. Sprawdzamy czy to wersja deweloperska np. 0.1.0-dev.4
	if let Some((base, dev_num_str)) = ver.rsplit_once("-dev.")
		&& let Ok(num) = dev_num_str.parse::<u64>()
	{
		return format!("{}-dev.{}", base, num + 1);
	}

	// 2. Jeśli to standardowe SemVer X.Y.Z -> podbijamy patch
	if let Some((major_minor, patch_str)) = ver.rsplit_once('.')
		&& let Ok(patch) = patch_str.parse::<u64>()
	{
		return format!("{}.{}", major_minor, patch + 1);
	}

	// Fallback
	format!("{}-dev.1", ver)
}

	}

use std::{fs, process::Command};

use inquire::Select;

const HISTORY_FILE: &str = ".cargo/.x-do-history.txt";

pub fn add_to_history(cmd: &str, args: &[&str]) {
	let _ = fs::create_dir_all(".cargo");
	let full_cmd = format!("{} {}", cmd, args.join(" "));

	let mut history = read_history();
	// Przenieś na górę, jeśli już istnieje
	history.retain(|x| x != &full_cmd);
	history.insert(0, full_cmd);

	// Zatrzymaj maksymalnie 20 ostatnich komend
	history.truncate(20);

	let _ = fs::write(HISTORY_FILE, history.join("\n"));
}

pub fn read_history() -> Vec<String> {
	if let Ok(content) = fs::read_to_string(HISTORY_FILE) {
		content.lines().map(|s| s.trim().to_string()).filter(|s| !s.is_empty()).collect()
	} else {
		Vec::new()
	}
}

pub fn has_history() -> bool { !read_history().is_empty() }

pub fn handle_last(is_dry_run: bool) -> bool {
	let history = read_history();
	if let Some(last) = history.first() {
		execute_from_str(last, is_dry_run);
	} else {
		println!("🚫 Brak historii poleceń.");
	}
	true
}

pub fn handle_list(is_dry_run: bool) -> bool {
	let history = read_history();
	if history.is_empty() {
		println!("🚫 Brak historii poleceń.");
		return true;
	}

	match Select::new("📜 Wybierz polecenie do ponownego wykonania:", history).prompt() {
		Ok(choice) => {
			execute_from_str(&choice, is_dry_run);
		}
		Err(_) => println!("Anulowano."),
	}
	true
}

fn execute_from_str(cmd_str: &str, is_dry_run: bool) {
	let mut parts = cmd_str.split_whitespace();
	if let Some(prog) = parts.next() {
		let args: Vec<&str> = parts.collect();
		if is_dry_run {
			println!("> [DRY RUN] 🔒 Uruchomiłbym: {} {}", prog, args.join(" "));
		} else {
			println!("> Wykonuję: {} {}\r\n", prog, args.join(" "));
			let _ = Command::new(prog).args(&args).status();
			// Zaktualizuj historię
			add_to_history(prog, &args);
		}
	}
}

fn detect_project_name() -> String {
	if let Ok(content) = fs::read_to_string("Cargo.toml") {
		for line in content.lines() {
			let trimmed = line.trim();
			if trimmed.starts_with("name")
				&& let Some((_, val)) = trimmed.split_once('=')
			{
				let name = val.trim().trim_matches('"').trim_matches('\'');
				if !name.is_empty() {
					return name.to_string();
				}
			}
		}
	}
	env::current_dir()
		.ok()
		.and_then(|p| p.file_name().map(|s| s.to_string_lossy().to_string()))
		.unwrap_or_else(|| "project".to_string())
}

fn load_config() -> SnapshotConfig {

		Path::new("./.cargo/.x-do.toml"),
		Path::new("./.cargo/x-do.toml"),
		Path::new("./.xtask/.x-do.toml"),
		Path::new("./.xtask/x-do.toml"),

	];

	for path in candidate_paths {
		if let Ok(content) = fs::read_to_string(path)
			&& let Ok(root) = toml::from_str::<RootConfig>(&content)

		}
	}

}

fn map_columns(cols: &[String]) -> Vec<Column> {
	cols.iter()
		.filter_map(|c| match c.as_str() {
			"Weight" => Some(Column::Weight),
			"Temporal" => Some(Column::Temporal),
			"Path" => Some(Column::Path),
			_ => None,
		})
		.collect()
}

fn build_sorting(cfg: &SnapshotConfig) -> Sorting {
	Sorting::new()
		.enabled(true)
		.group_strategy(GroupStrategy::Custom(vec![NodeGroup::TextFile, NodeGroup::Directory, NodeGroup::BinaryFile]))
		.same_name_priority(match cfg.sort.same_name_priority.as_str() {
			"DirectoryFirst" => SameNamePriority::DirectoryFirst,
			_ => SameNamePriority::FileFirst,
		})
		.no_ext_priority(match cfg.sort.no_ext_priority.as_str() {
			"Below" => NoExtPriority::Below,
			_ => NoExtPriority::Above,
		})
		.ignore_leading_dot(cfg.sort.ignore_leading_dot)
		.char_class_order([CharClass::Special, CharClass::Digit, CharClass::Letter])
		.case_precedence(match cfg.sort.case_precedence.as_str() {
			"UpperFirst" => CasePrecedence::UpperFirst,
			"LowerFirst" => CasePrecedence::LowerFirst,
			_ => CasePrecedence::Insensitive,
		})
}

fn build_bundle(cfg: &SnapshotConfig, sorting: Sorting) -> QueryPathBundle {
	QueryPathBundle::new()
		.queueing_before_numerate(sorting)
		.numerate_dirs(cfg.incl.numerate_dirs)
		.numerate_binaries(cfg.incl.numerate_binaries)
		.numerate_skip(&cfg.incl.numerate_skip)
		.numerate_start_from(cfg.incl.numerate_start_from)
}

fn build_fmt(cfg: &SnapshotConfig) -> QueryPathFmt {
	let weight_display = match cfg.tree.weight.dir_display.as_str() {
		"None" => DirWeightDisplay::None,
		"RealOnly" => DirWeightDisplay::RealOnly,
		"Both" => DirWeightDisplay::Both,
		_ => DirWeightDisplay::MatchedOnly,
	};

	let weight_precision = match cfg.tree.weight.precision.as_str() {
		"0.01" | "Hundredths" => WeightPrecision::Hundredths,
		_ => WeightPrecision::Tenths,
	};

	let unit_system = if cfg.tree.weight.unit_system_binary { UnitSystem::Binary } else { UnitSystem::Decimal };

	QueryPathFmt::new()
		.name_width(cfg.tree.name_width)
		.path_width(cfg.tree.path_width)
		.column_order_left(map_columns(&cfg.tree.column_order_left))
		.column_order_right(map_columns(&cfg.tree.column_order_right))
		.stats_weight(
			StatsWeight::new()
				.enabled(cfg.tree.weight.stats_weight)
				.unit_system(unit_system)
				.dir_display(weight_display)
				.precision(weight_precision),
		)
		.stats_temporal(
			StatsTemporal::new().enabled(cfg.tree.temporal.stats_temporal).pattern(&cfg.tree.temporal.pattern),
		)
}

	if final_name.contains("project") {

	let mut snapshot = Snapshot::new()
		.title(&final_title)
		.name(&final_name)
		.output_dir(&cfg.output_dir)
		.max_single_file_size(cfg.max_single_file_size)
		.content_skip(&cfg.incl.content_skip)

		.add_footer_customize_text(&cfg.add_fotter_customize_text);

	if cfg.title_sufix_date {
		snapshot = snapshot.title_suffix_date_fmt(&cfg.title_sufix_date_fmt);
	}

	if cfg.name_sufix_date {
		snapshot = snapshot.name_suffix_date_fmt(&cfg.name_sufix_date_fmt);
	}

	match snapshot.generate_and_save(&results, &idn, &fmt) {
		Ok(path) => println!("✅ Zapisano snapshot do: {}\r\n", path.display()),
		Err(e) => eprintln!("❌ Błąd tworzenia migawki: {}\r\n", e),
	}
}
